# Tree uses arrays for object members, so even the independent final serializer
# needs only ASCII wrapper-key sorting. UTF16 member sorting is explicit below.
def tree(value):
    t = type(value)
    if value is None:
        return ['null']
    if t is bool:
        return ['bool', value]
    if t is int:
        return ['number', value, 1]
    if t is float:
        if not math.isfinite(value):
            raise ValueError('nonfinite')
        n, d = value.as_integer_ratio()
        return ['number', n, d]
    if t is str:
        value.encode('utf-8')
        return ['string', value]
    if t is list:
        return ['array', [tree(item) for item in value]]
    if t is dict:
        if not all(type(key) is str for key in value):
            raise ValueError('non-string key')
        for key in value:
            key.encode('utf-8')
        return ['object', [[key, tree(value[key])] for key in sorted(value, key=lambda key: key.encode('utf-16-be'))]]
    raise ValueError('unsupported type')

def material(value, profile='review-vector-v1'):
    wrapped = {'codec': 'vpwiki.runtime-tree.v1', 'profile': profile, 'tree': tree(value)}
    independent = json.dumps(wrapped, ensure_ascii=False, sort_keys=True, separators=(',', ':'), allow_nan=False).encode('utf-8')
    assert independent == canonicalize(wrapped)
    return independent

vectors = {}
for name, value in [('null', None), ('true', True), ('one_int', 1), ('one_float', 1.0), ('zero_int', 0), ('negative_zero', -0.0), ('point_one', 0.1), ('next_one', math.nextafter(1.0, math.inf)), ('k1', 1.5), ('b', 0.75), ('subnormal', math.ulp(0.0)), ('max_float', float.fromhex('0x1.fffffffffffffp+1023')), ('tag_like_array', ['number', 1, 1]), ('utf16_order', {'\ue000': 1, '\U00010000': 2})]:
    data = material(value)
    vectors[name] = {'canonical_utf8': data.decode(), 'sha256': hashlib.sha256(data).hexdigest()}
assert vectors['one_int'] == vectors['one_float']
assert vectors['zero_int'] == vectors['negative_zero']
assert vectors['true'] != vectors['one_int']
assert vectors['one_float'] != vectors['next_one']
assert vectors['tag_like_array'] != vectors['one_int']
assert tree({'\ue000': 1, '\U00010000': 2})[1][0][0] == '\U00010000'
for invalid in (float('nan'), float('inf'), -float('inf'), '\ud800', {1: 'x'}, (1, 2)):
    try:
        material(invalid)
    except (ValueError, UnicodeError):
        pass
    else:
        raise AssertionError(repr(invalid))
